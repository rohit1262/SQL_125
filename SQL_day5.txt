24.. List the largest orders for October 3, for each salesperson

select o.onum,o.amt,o.odate,o.snum,s.sname from orders o join salespeople s on o.snum = s.snum where  o.odate = '1996-10-03' and o.amt =( select max(o2.amt) from orders o2 where o2.snum = o.snum and o2.odate = o.odate);
